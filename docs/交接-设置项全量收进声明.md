```js
const vm = require('vm');
const src = read('web/ui/ui.js');
/* 从 `window.UI_PREFS = {` 起做括号扫描，跳过注释与字符串，取出整个对象字面量 */
function extractObject(src, marker) {
  const at = src.indexOf(marker);
  if (at < 0) return '';
  let i = src.indexOf('{', at), depth = 0, out = '';
  for (; i < src.length; i++) {
    const c = src[i], n = src[i + 1];
    if (c === '/' && n === '/') { const j = src.indexOf('\n', i); i = j < 0 ? src.length : j; out += '\n'; continue; }
    if (c === '/' && n === '*') { const j = src.indexOf('*/', i + 2); i = j < 0 ? src.length : j + 1; out += ' '; continue; }
    if (c === "'" || c === '"' || c === '`') {
      const q = c; out += c;
      for (i++; i < src.length; i++) {
        const d = src[i]; out += d;
        if (d === '\\') { i++; out += src[i]; continue; }
        if (d === q) break;
      }
      continue;
    }
    if (c === '{') depth++;
    if (c === '}') { depth--; if (!depth) { out += c; break; } }
    out += c;
  }
  return out;
}
const sandbox = { window: { VIEW_MODES: [['list', '列表'], ['compact', '紧凑']] },
  localStorage: { getItem: () => null }, console };
vm.runInNewContext('window.UI_PREFS = ' + extractObject(src, 'window.UI_PREFS = {') + ';', sandbox);
const UP = sandbox.window.UI_PREFS;
```

## 6. 坑（踩过或推演出来的）

1. **先认节、再套默认、最后看显式标记** —— 反过来 `ai_chat_off` 的 `export:1` 会被
   `ai_chat_` 前缀的 `export:0` 截走。
2. **`Net.NUMS` 的名字 ≠ 存储键**：`timeout` ↔ `sk_net_timeout`。`data-mkey` 用存储键，
   `Net.setNum` 用名字；`sk_poll_sec` 不在 `sk_net_` 前缀下，所以它的 setRow 也带 `num`。
3. **`sb_bal_cache` 不能被 `sb_bal_` 前缀带进去** —— families 只有精确键 `sb_bal_show`。
4. **`sk_model_meta_` 必须压过 `sk_model_`** —— `markOf` 的前缀按最长优先。
5. **`dir_dock_h` / `dir_ws_unload` 的精确键必须排在 `dir_` 前缀之前被认出来** ——
   精确键本来就先查，保持声明里有这两条即可。
6. **`_cache_v1` 是后缀，后缀先于前缀** —— 所以 `audio_tasks_cache_v1` 不会被 `audio_tasks` 带同步。
7. **`UI_PREFS` 的方法依赖 `this`**，测试里不要解构；直接 `UP.syncOf(k)`。
8. **`flagOf` 的显式 `0` 就是 false**，`null/undefined` 才落到节默认。
9. **`assistSpecs` 会跳过 `assist:'page'|'chars'|'model'` 的族**，动态三族由调用方拼。
10. **`settings.js` 的 `SET_ROWS` 在模块求值期读 `window.UI_PREFS`**：ui.js 是壳里的同步经典脚本，
    设置页按需加载，顺序有保证；声明缺失时宁可空表（与 datasync 同一策略）。
11. **`LEGACY_KEYS` 里那些旧键**（`sk_text_provider`、`char_opts`、`sk_autosave`、`drama_chars`…）
    不登记进声明是对的：它们开机就被连镜像一起删，不需要同步 / 导出待遇。
12. **`web/ui/datasync.js` 的 `workspace`** 走 `mirrorWs`，不要给它补 `syncOf` 声明。
13. **G23 注释规则**：新写的注释只解释功能，不写文档名 / 章节号 / 任务号。
